! =====================================================================
! Thiet bi : SW-HQ
! Model    : Cisco 2960-24TT
! Vai tro  : Switch LAN, uplink G0/1 den R-HQ G0/0
! Nguoi cau hinh : TV1 - Le Minh Dao (Nhom 02)
! Cach dung: mo CLI cua thiet bi o che do Router>/Switch>, dan toan bo file.
! Ghi chu  : mat khau cisco/class chi dung cho mo phong.
! =====================================================================
enable
configure terminal
!
! ---------- 1. Thong tin chung ----------
hostname SW-HQ
service password-encryption
enable secret class
no ip domain-lookup
!
! ---------- 2a. Cong ket noi ----------
interface FastEthernet0/1
 description Server-TT
 switchport mode access
!
interface FastEthernet0/2
 description Server-Backup
 switchport mode access
!
interface FastEthernet0/3
 description PC-Admin
 switchport mode access
!
interface range FastEthernet0/4 - 24
 description Khong su dung
 shutdown
!
interface GigabitEthernet0/1
 description Uplink den R-HQ G0/0
!
interface GigabitEthernet0/2
 description Khong su dung
 shutdown
!
! ---------- 2b. IP quan tri ----------
interface Vlan1
 description Quan tri switch
 ip address 192.168.0.2 255.255.255.0
 no shutdown
!
ip default-gateway 192.168.0.1
!
! ---------- 3. Bao mat truy cap thiet bi ----------
! Gioi han nguon Telnet (access-list 10 + access-class tren VTY)
! do TV2 cau hinh trong Nhom02_v2.pkt theo bang phan quyen muc 2.5
!
banner motd #SW-HQ - Chi danh cho quan tri vien duoc phep truy cap#
!
line console 0
 password cisco
 login
 exec-timeout 10 0
!
line vty 0 4
 password cisco
 login
 transport input telnet
 exec-timeout 10 0
!
line vty 5 15
 password cisco
 login
 transport input telnet
 exec-timeout 10 0
!
end
write memory
