! ===== SW-CH3 – phần TV2 bổ sung: SSH thay Telnet, port security, NTP, Syslog =====
! Dán vào CLI sau khi đăng nhập (console: cisco). Lệnh enable hỏi mật khẩu: class
! Làm sau file SW-CH3_VTY.txt (ACL / access-class).
enable
configure terminal
! SSH: tài khoản riêng + tên miền + khóa RSA (Packet Tracer không lưu sẵn khóa trong file)
username admin privilege 15 secret cisco
ip domain-name nhom02.local
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line vty 0 15
 login local
 transport input ssh
 exit
! Port security: mỗi cổng chỉ nhận đúng 1 máy (ghi nhớ MAC sticky), máy lạ cắm vào -> tắt cổng;
! bpduguard: cắm switch lạ vào cổng máy trạm cũng bị tắt cổng
interface range fa0/1 - 5
 switchport mode access
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 exit
! Cổng không dùng: tắt hẳn
interface range fa0/6 - 24
 shutdown
 exit
interface g0/2
 shutdown
 exit
! NTP: lấy giờ từ Server-Backup (Services > NTP: On)
ntp server 192.168.0.11
! Syslog: gửi nhật ký kèm giờ về Server-Backup (Services > SYSLOG: On)
service timestamps log datetime msec
logging 192.168.0.11
end
write memory
! Sau khi dán: cho từng máy trong cửa hàng 3 ping gateway 1 lần để switch học MAC sticky.
