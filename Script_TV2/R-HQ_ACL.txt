! ===== R-HQ – phần TV2: ACL bảo vệ vùng máy chủ (áp OUT trên G0/0) + giới hạn telnet =====
! Dán vào CLI sau khi đăng nhập (console: cisco). Lệnh enable hỏi mật khẩu: class
enable
configure terminal
! 1) Khai báo ACL trước, áp sau
ip access-list extended HQ_SERVERS_OUT
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.0.10 eq www
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.0.10 eq 443
 permit icmp 192.168.10.0 0.0.0.255 host 192.168.0.10
 permit tcp 192.168.20.0 0.0.0.255 host 192.168.0.10 eq www
 permit tcp 192.168.20.0 0.0.0.255 host 192.168.0.10 eq 443
 permit icmp 192.168.20.0 0.0.0.255 host 192.168.0.10
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.0.10 eq www
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.0.10 eq 443
 permit icmp 192.168.30.0 0.0.0.255 host 192.168.0.10
 permit udp host 192.168.10.2 host 192.168.0.11
 permit udp host 192.168.20.2 host 192.168.0.11
 permit udp host 192.168.30.2 host 192.168.0.11
 permit udp 10.0.0.0 0.0.255.255 host 192.168.0.11
 permit icmp 10.0.0.0 0.0.255.255 host 192.168.0.10
 permit icmp any host 192.168.0.20
 permit tcp any host 192.168.0.20 established
 deny ip any any
 exit
! 2) Áp ACL vào cổng LAN đúng hướng
interface GigabitEthernet0/0
 ip access-group HQ_SERVERS_OUT out
 exit
! 3) Chỉ PC-Admin 192.168.0.20 được telnet vào router
access-list 10 permit host 192.168.0.20
line vty 0 4
 access-class 10 in
 exit
end
write memory
