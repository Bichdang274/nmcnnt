! Vai tro  : Switch LAN, uplink G0/1 den R-CH3 G0/0

! Cach dung: mo CLI cua thiet bi o che do Router>/Switch>, dan toan bo file.
! Ghi chu  : mat khau cisco/class chi dung cho mo phong.

enable
configure terminal
!
! ---------- 1. Thong tin chung ----------
hostname SW-CH3
service password-encryption
enable secret class
no ip domain-lookup
!
! ---------- 2a. Cong ket noi ----------
interface FastEthernet0/1
 description POS3-1
 switchport mode access
!
interface FastEthernet0/2
 description POS3-2
 switchport mode access
!
interface FastEthernet0/3
 description POS3-3
 switchport mode access
!
interface FastEthernet0/4
 description POS3-4
 switchport mode access
!
interface FastEthernet0/5
 description POS3-5
 switchport mode access
!
interface range FastEthernet0/6 - 24
 description Khong su dung
 shutdown
!
interface GigabitEthernet0/1
 description Uplink den R-CH3 G0/0
!
interface GigabitEthernet0/2
 description Khong su dung
 shutdown
!
! ---------- 2b. IP quan tri ----------
interface Vlan1
 description Quan tri switch
 ip address 192.168.30.2 255.255.255.0
 no shutdown
!
ip default-gateway 192.168.30.1
!
! ---------- 3. Bao mat truy cap thiet bi ----------
! Chi PC-Admin (192.168.0.20) duoc Telnet vao thiet bi
access-list 10 permit host 192.168.0.20
!
banner motd #SW-CH3 - Chi danh cho quan tri vien duoc phep truy cap#
!
line console 0
 password cisco
 login
 exec-timeout 10 0
!
line vty 0 4
 access-class 10 in
 password cisco
 login
 transport input telnet
 exec-timeout 10 0
!
line vty 5 15
 access-class 10 in
 password cisco
 login
 transport input telnet
 exec-timeout 10 0
!
end
write memory
